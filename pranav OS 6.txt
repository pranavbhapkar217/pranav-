

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <sys/wait.h>
#include <ctype.h>

int main()
{
    int fd1, fd2;
    char buffer[4096];
    int chars = 0, words = 0, lines = 0;
    int inword = 0, n, i, last = '\n';

    mkfifo("fifo1", 0666);
    mkfifo("fifo2", 0666);

    if (fork() == 0)
    {
        fd1 = open("fifo1", O_RDONLY);

        FILE *fp = fopen("output.txt", "w");

        while ((n = read(fd1, buffer, sizeof(buffer))) > 0)
        {
            for (i = 0; i < n; i++)
            {
                chars++;

                if (buffer[i] == '\n')
                    lines++;

                if (isspace((unsigned char)buffer[i]))
                    inword = 0;
                else if (!inword)
                {
                    words++;
                    inword = 1;
                }

                last = (unsigned char)buffer[i];
            }
        }

        if (chars > 0 && last != '\n')
            lines++;

        close(fd1);

        fprintf(fp, "Number of Characters: %d\n", chars);
        fprintf(fp, "Number of Words: %d\n", words);
        fprintf(fp, "Number of Lines: %d\n", lines);

        fclose(fp);

        fp = fopen("output.txt", "r");
        fd2 = open("fifo2", O_WRONLY);

        while ((n = fread(buffer, 1, sizeof(buffer), fp)) > 0)
        {
            write(fd2, buffer, n);
        }

        fclose(fp);
        close(fd2);

        exit(0);
    }
    else
    {
        printf("Enter sentences (type END to finish):\n");

        fd1 = open("fifo1", O_WRONLY);

        while (1)
        {
            if (fgets(buffer, sizeof(buffer), stdin) == NULL)
                break;

            if (strcmp(buffer, "END\n") == 0)
                break;

            write(fd1, buffer, strlen(buffer));
        }

        close(fd1);

        wait(NULL);

        fd2 = open("fifo2", O_RDONLY);

        printf("\nResult received from Process 2:\n");

        while ((n = read(fd2, buffer, sizeof(buffer) - 1)) > 0)
        {
            buffer[n] = '\0';
            printf("%s", buffer);
        }

        close(fd2);
    }

    unlink("fifo1");
    unlink("fifo2");

    return 0;
}
